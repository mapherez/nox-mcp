package consumer

import (
	"bytes"
	"context"
	"encoding/json"
	"github.com/google/jsonschema-go/jsonschema"
	nox "github.com/mapherez/nox-mcp/go"
	"github.com/modelcontextprotocol/go-sdk/mcp"
	"net/http"
	"net/http/httptest"
	"testing"
)

func TestIndependentConsumer(t *testing.T) {
	executed := 0
	schema := &jsonschema.Schema{Type: "object", Properties: map[string]*jsonschema.Schema{"value": {Type: "string"}}, Required: []string{"value"}, AdditionalProperties: &jsonschema.Schema{Not: &jsonschema.Schema{}}}
	runtime, err := nox.New(nox.Options{AppID: "consumer", Name: "consumer", Version: "0.5.0", Tools: []nox.Tool{{Name: "echo", Description: "Consumer regression", InputSchema: schema, OutputSchema: schema, RequiredScopes: []string{"write"}, Execute: func(_ nox.ExecutionContext, input map[string]any) (map[string]any, error) {
		executed++
		return input, nil
	}}}})
	if err != nil {
		t.Fatal(err)
	}
	server := httptest.NewServer(runtime.HTTPHandler())
	defer server.Close()
	for _, modern := range []bool{false, true} {
		params := map[string]any{"name": "echo", "arguments": map[string]any{"value": "hello"}}
		if modern {
			params["_meta"] = map[string]any{"io.modelcontextprotocol/protocolVersion": "2026-07-28", "io.modelcontextprotocol/clientCapabilities": map[string]any{}}
		}
		data, _ := json.Marshal(map[string]any{"jsonrpc": "2.0", "id": 1, "method": "tools/call", "params": params})
		request, _ := http.NewRequestWithContext(context.Background(), "POST", server.URL, bytes.NewReader(data))
		request.Header.Set("Content-Type", "application/json")
		request.Header.Set("Accept", "application/json, text/event-stream")
		if modern {
			request.Header.Set("Mcp-Protocol-Version", "2026-07-28")
			request.Header.Set("Mcp-Method", "tools/call")
			request.Header.Set("Mcp-Name", "echo")
		}
		response, err := http.DefaultClient.Do(request)
		if err != nil {
			t.Fatal(err)
		}
		var wire struct {
			Result mcp.CallToolResult
			Error  any
		}
		err = json.NewDecoder(response.Body).Decode(&wire)
		response.Body.Close()
		if err != nil {
			t.Fatal(err)
		}
		if wire.Error != nil || !wire.Result.IsError || wire.Result.StructuredContent != nil {
			t.Fatalf("invalid error envelope: %#v", wire)
		}
		var body nox.Error
		if err := json.Unmarshal([]byte(wire.Result.Content[0].(*mcp.TextContent).Text), &body); err != nil {
			t.Fatal(err)
		}
		if body.Code != "FORBIDDEN" || executed != 0 {
			t.Fatalf("authorization lost: %#v executions=%d", body, executed)
		}
	}
	if _, err := nox.Fingerprint(map[string]any{"x": 1}); err != nil {
		t.Fatal(err)
	}
}
